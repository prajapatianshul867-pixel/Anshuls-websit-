from http.server import HTTPServer, BaseHTTPRequestHandler html = """ Anshul Website

✦ ANSHUL WEBSITE ✦

Home About Contact

Welcome to My Website!

Namaste dosto! Meri colorful website mein aapka swagat hai.

🌟 About Me

Welcome to my creative world!

Yahan coding, creativity aur technology ka magic milega.

📩 Contact

Mujhse judne ke liye button dabao.

© 2026 Anshul Website | Made with ❤️
""" class Website(BaseHTTPRequestHandler): def do_GET(self): self.send_response(200) self.send_header("Content-type", "text/html; charset=utf-8") self.end_headers() self.wfile.write(html.encode("utf-8")) server = HTTPServer(("0.0.0.0", 8000), Website) print("Website running!") print("Open your browser: http://127.0.0.1:8000") server.serve_forever()
